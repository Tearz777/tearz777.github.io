Core Formula
Dua Formula Kunci

SKU auto-generate dan order tracking — dua formula yang mengikat seluruh arsitektur 3 pilar.

1 / 2 — Transaksi!E11 · Kode SKU
ARRAY_CONSTRAIN + IFS + XLOOKUP
=ARRAY_CONSTRAIN(ARRAYFORMULA( IFS( // Kondisi 1 — sel kosong, skip B11="", "", // Kondisi 2 — tidak ada di Master → slug COUNTIF('Master Barang'!B:B, B11)=0, SUBSTITUTE(TRIM(B11), " ", "-"), // Kondisi 3 — ditemukan → ambil SKU dari master COUNTIF('Master Barang'!B:B, B11)=1, XLOOKUP(B11, 'Master Barang'!B:B, 'Master Barang'!C:C,,,1) ) ), 1, 1)
2 / 2 — Jadwal!I2 · Keterangan
LET + FILTER + IFERROR + INDEX
=IF(OR(B2="", E2=""), "", LET( Nama, B2, SKU, C2, Estimasi, E2, QtyOrder, F2, // Cross-reference ke sheet Transaksi FilterTrans, FILTER( Transaksi!B:H, (Transaksi!F:F="Barang Datang") * (Transaksi!D:D=Nama) * (Transaksi!E:E=SKU) ), TotalDatang, IFERROR(SUM(INDEX(FilterTrans,,7)), 0), TanggalDatang, IFERROR(MAX(INDEX(FilterTrans,,1)), 0), SelisihQty, QtyOrder-TotalDatang, SelisihHari, TanggalDatang-Estimasi, KetQty, IF(SelisihQty=0, "Oke", IF(SelisihQty>0, "Kurang "&SelisihQty&" pcs", "Lebih "&ABS(SelisihQty)&" pcs")), KetHari, IF(TanggalDatang=0, "Belum datang", IF(SelisihHari=0, "", IF(SelisihHari>0, "Terlambat "&SelisihHari&" hari", "Lebih cepat "&ABS(SelisihHari)&" hari"))), Hasil, IF(TanggalDatang=0, KetHari, IF(AND(SelisihQty=0, SelisihHari=0), "Oke", IF(SelisihQty=0, KetHari, IF(SelisihHari=0, KetQty, KetHari&" dengan "&LOWER(KetQty))))), Hasil ))
